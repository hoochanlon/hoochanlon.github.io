{{- $text := .Text | strings.TrimSpace -}} {{- $lines := split $text "\n" -}} {{- $firstLine := index $lines 0 | strings.TrimSpace -}} {{- $alertType := "" -}} {{- $alertIcon := "" -}} {{- $alertTitle := "" -}} {{- if strings.HasPrefix $firstLine "[!NOTE]" -}} {{- $alertType = "note" -}} {{- $alertIcon = "circle-info" -}} {{- $alertTitle = "Note" -}} {{- else if strings.HasPrefix $firstLine "[!TIP]" -}} {{- $alertType = "tip" -}} {{- $alertIcon = "lightbulb" -}} {{- $alertTitle = "Tip" -}} {{- else if strings.HasPrefix $firstLine "[!IMPORTANT]" -}} {{- $alertType = "important" -}} {{- $alertIcon = "comment" -}} {{- $alertTitle = "Important" -}} {{- else if strings.HasPrefix $firstLine "[!WARNING]" -}} {{- $alertType = "warning" -}} {{- $alertIcon = "triangle-exclamation" -}} {{- $alertTitle = "Warning" -}} {{- else if strings.HasPrefix $firstLine "[!CAUTION]" -}} {{- $alertType = "caution" -}} {{- $alertIcon = "xmark" -}} {{- $alertTitle = "Caution" -}} {{- end -}} {{- if $alertType -}} {{- $content := strings.TrimPrefix (printf "[!%s]" (upper $alertType)) $firstLine | strings.TrimSpace -}} {{- if gt (len $lines) 1 -}} {{- $restLines := after 1 $lines -}} {{- $content = printf "%s\n%s" $content (delimit $restLines "\n") -}} {{- end -}}
{{- partial "icon.html" $alertIcon -}} {{ $alertTitle }}
{{- $content | markdownify -}}
{{- else -}}
{{- .Text | safeHTML -}}
{{- end -}}
