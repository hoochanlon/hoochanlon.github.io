{{- /* title-display.html: 接收一个字符串参数，按 标记强制换行 - 含 : 按其分割，每段包 whitespace-nowrap，段间插
- 不含 : 原样输出 - 调用方负责传入最终要显示的文本（已从 frontmatter 取好字段） */ -}} {{- $text := . | emojify -}} {{- if in $text "" -}} {{- $parts := split $text "" -}} {{- $last := sub (len $parts) 1 -}} {{- range $idx, $part := $parts -}} {{- $part | safeHTML -}}{{- if lt $idx $last -}}
{{- end -}} {{- end -}} {{- else -}} {{- $text | safeHTML -}} {{- end -}}
